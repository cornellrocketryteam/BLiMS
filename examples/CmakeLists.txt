cmake_minimum_required(VERSION 3.22)
include(../lib/pico-sdk/pico_sdk_init.cmake)
project(examples C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
add_compile_options(-Wall -Wno-format -Wno-unused-function)

# i2c_scan
add_executable(i2c_scan i2c_scan.cpp)
target_link_libraries(i2c_scan pico_stdlib hardware_gpio hardware_i2c)
pico_enable_stdio_usb(i2c_scan 1)
pico_enable_stdio_uart(i2c_scan 0)
pico_add_uf2_output(i2c_scan)

# controller_test
add_executable(controller_test controller_test.cpp)
target_include_directories(controller_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../lib/ublox-MX-Pico)
add_subdirectory(../lib/ublox-MX-Pico ublox-MX-Pico)
target_link_libraries(controller_test pico_stdlib hardware_gpio hardware_i2c hardware_pwm hardware_timer ublox-MX-Pico)
pico_enable_stdio_usb(controller_test 1)
pico_enable_stdio_uart(controller_test 0)
pico_add_uf2_output(controller_test)
