cmake_minimum_required(VERSION 3.27)

# We only need CXX (C++) for this test
project(blims_sim CXX) 

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Standard warnings to help you debug
add_compile_options(
    -Wall
    -Wextra
)

# Add your source file
set(SOURCES
  blims_sim.cpp
)

# Create the standard executable for your Mac
add_executable(${PROJECT_NAME} ${SOURCES})
